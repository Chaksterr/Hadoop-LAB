HADOOP-LAB - MANUAL COMMANDS
Run on master as hadoop:  vagrant ssh master  ->  sudo -iu hadoop


START  (ZooKeeper -> HDFS -> YARN -> Hive -> HBase)
-----
0. for n in master datanode1 datanode2; do ssh $n '~/zookeeper/bin/zkServer.sh start'; done
1. start-dfs.sh
2. start-yarn.sh
3. mapred --daemon start historyserver
4. nohup ~/hive/bin/hive --service metastore > ~/hive-metastore.log 2>&1 &
5. sleep 10
6. nohup ~/hive/bin/hive --service hiveserver2 > ~/hiveserver2.log 2>&1 &
7. hdfs dfsadmin -safemode wait
8. start-hbase.sh


STOP  (HBase -> Hive -> YARN -> HDFS -> ZooKeeper)
----
1. hbase master stop --shutDownCluster      (not stop-hbase.sh, it hangs)
2. jps                                      (repeat until HMaster is gone, on master AND datanode1)
3. (ZooKeeper is stopped last, see step 9)
4. pkill -f 'org.apache.hive.service.server.HiveServer2'
5. pkill -f 'hive.*metastore'
6. mapred --daemon stop historyserver
7. stop-yarn.sh
8. stop-dfs.sh
9. for n in master datanode1 datanode2; do ssh $n '~/zookeeper/bin/zkServer.sh stop'; done


KAFKA LAB  (ZooKeeper -> Kafka ; no HDFS/HBase needed)
---------
1. for n in master datanode1 datanode2; do ssh $n '~/zookeeper/bin/zkServer.sh start'; done
2. for n in master datanode1 datanode2; do ssh $n 'export KAFKA_HEAP_OPTS="-Xmx512M -Xms512M"; ~/kafka/bin/kafka-server-start.sh -daemon ~/kafka/config/server.properties'; done
3. ~/zookeeper/bin/zkCli.sh -server master:2181 ls /kafka/brokers/ids          (expect [1, 2, 3])
4. ~/kafka/bin/kafka-topics.sh --create --if-not-exists --topic INGTopic --partitions 3 --replication-factor 2 --bootstrap-server master:9092
5. ~/kafka/bin/kafka-topics.sh --describe --topic INGTopic --bootstrap-server master:9092
6. jupyter notebook --ip=0.0.0.0 --port=8888 --no-browser --notebook-dir=~/notebooks   ->  http://192.168.56.10:8888
STOP: for n in master datanode1 datanode2; do ssh $n '~/kafka/bin/kafka-server-stop.sh'; done
      then stop ZooKeeper on the 3 nodes (zkServer.sh stop)
